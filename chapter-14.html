m, n = cost.shape
A_eq, b_eq = [], []
for i in range(m):
    row = np.zeros(m * n)
    row[i * n:(i + 1) * n] = 1
    A_eq.append(row); b_eq.append(supply[i])

    A_eq.append(row); b_eq.append(demand[j])
result = linprog(cost.ravel(), A_eq=np.array(A_eq),
                 b_eq=np.array(b_eq), bounds=(0, None), method="highs")
print(result.x.reshape(m, n), result.fun)

习题 04下面给出Dijkstra算法及一个有向图

下面给出 Dijkstra 算法及一个有向图。将图改为无向图，并分别以各节点为源点计算最短距离。

import heapq
def dijkstra(graph, source):
    distance = {v: float("inf") for v in graph}
    distance[source], heap = 0, [(0, source)]
    while heap:
        dist, u = heapq.heappop(heap)
        if dist > distance[u]: continue
        for v, weight in graph[u].items():
            new_dist = dist + weight
            if new_dist < distance[v]:
                distance[v] = new_dist
                heapq.heappush(heap, (new_dist, v))
    return distance

习题 05下面的动态规划程序求解0–1背包问题，并…

下面的动态规划程序求解 0–1 背包问题，并返回所选物品的 Python 下标。

import numpy as np
def knapsack_01(weights, values, capacity):
    n = len(weights)
    dp = np.zeros((n + 1, capacity + 1), dtype=int)
    keep = np.zeros((n + 1, capacity + 1), dtype=bool)
    for i in range(1, n + 1):
        w, v = weights[i - 1], values[i - 1]
        for cap in range(capacity + 1):
            dp[i, cap] = dp[i - 1, cap]
            if w <= cap and dp[i-1, cap-w] + v > dp[i, cap]:
                dp[i, cap] = dp[i-1, cap-w] + v
                keep[i, cap] = True
    selected, cap = [], capacity
    for i in range(n, 0, -1):
        if keep[i, cap]:
            selected.append(i - 1); cap -= weights[i - 1]
    return dp[n, capacity], list(reversed(selected))

令重量 ，价值 ，分别取容量 ，比较最优选择和最大价值。

习题 06下面的函数计算公式系统指标

下面的函数计算 系统指标。取 ，令 ，用 try/except 记录不稳定情形，并比较稳定情形的 。

import math
def mmc(lam, mu, c):
    if lam <= 0 or mu <= 0:
        raise ValueError("lambda 和 mu 必须为正数")
    if not isinstance(c, int) or c <= 0:
        raise ValueError("c 必须为正整数")
    rho = lam / (c * mu)
    if rho >= 1:
        raise ValueError("系统不稳定：lambda 必须小于 c * mu")
    a = lam / mu
    P0 = 1 / (sum(a**k / math.factorial(k) for k in range(c))
              + a**c / (math.factorial(c) * (1-rho)))
    Lq = P0 * a**c * rho / (math.factorial(c) * (1-rho)**2)
    return {"rho":rho, "P0":P0, "Lq":Lq, "L":Lq+a,
            "Wq":Lq/lam, "W":(Lq+a)/lam}

习题 08下列程序按期望收益准则选择方案

下列程序按期望收益准则选择方案。两个状态分别为市场好和市场差。

alternatives = {
 "新建生产线": [(0.6, 120), (0.4, -30)],
 "外包生产":   [(0.6,  60), (0.4,  20)],
 "暂缓投资":   [(0.6,   0), (0.4,   0)]}
def expected_value(outcomes):
    return sum(p * value for p, value in outcomes)
for name, outcomes in alternatives.items():
    print(name, expected_value(outcomes))
best = max(alternatives, key=lambda x: expected_value(alternatives[x]))
print("推荐方案:", best)
